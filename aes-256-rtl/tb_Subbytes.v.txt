`timescale 1ns/1ps

module tb_subbytes;

    // Test signals
    reg [127:0] test_state;
    wire [127:0] subbytes_output;
    
    // Instantiate SubBytes module
    SubBytes sb_inst (
        .state_in(test_state),
        .state_out(subbytes_output)
    );
    
    initial begin
        $display("========== AES SUBBYTES TESTBENCH ==========\n");
        
        // Wait for S-box initialization
        #1;
        
        // TEST 1: Known Vector
        $display("TEST 1: Known Test Vector");
        test_state = 128'h00112233445566778899aabbccddeeff;
        $display("Input State:  %h", test_state);
        
        #2;
        
        $display("Output State: %h", subbytes_output);
        $display("Expected:     63ca9fa3f2f5c6b6a89fb5c6b79f5f8a");
        
        // Verify first few bytes
        $display("\nByte-by-Byte Verification (first 4 bytes):");
        $display("  Input[0]=0x00 -> Output[0]=0x%h (Expected: 0x63)", subbytes_output[7:0]);
        $display("  Input[1]=0x11 -> Output[1]=0x%h (Expected: 0xca)", subbytes_output[15:8]);
        $display("  Input[2]=0x22 -> Output[2]=0x%h (Expected: 0x9f)", subbytes_output[23:16]);
        $display("  Input[3]=0x33 -> Output[3]=0x%h (Expected: 0xa3)", subbytes_output[31:24]);
        
        // TEST 2: All Zeros
        $display("\n\nTEST 2: All Zeros State");
        test_state = 128'h00000000000000000000000000000000;
        $display("Input State:  %h", test_state);
        
        #2;
        
        $display("Output State: %h", subbytes_output);
        $display("Expected:     63636363636363636363636363636363");
        $display("Explanation:  sbox[0x00] = 0x63 for all 16 bytes");
        
        if (subbytes_output == 128'h63636363636363636363636363636363)
            $display("? TEST 2 PASSED");
        else
            $display("? TEST 2 FAILED");
        
        // TEST 3: All 0xFF
        $display("\n\nTEST 3: All 0xFF State");
        test_state = 128'hffffffffffffffffffffffffffffffff;
        $display("Input State:  %h", test_state);
        
        #2;
        
        $display("Output State: %h", subbytes_output);
        $display("Expected:     16161616161616161616161616161616");
        $display("Explanation:  sbox[0xff] = 0x16 for all 16 bytes");
        
        if (subbytes_output == 128'h16161616161616161616161616161616)
            $display("? TEST 3 PASSED");
        else
            $display("? TEST 3 FAILED");
        
        // TEST 4: Sequential Pattern
        $display("\n\nTEST 4: Sequential Pattern 0x00-0x0F");
        test_state = 128'h000102030405060708090a0b0c0d0e0f;
        $display("Input State:  %h", test_state);
        
        #2;
        
        $display("Output State: %h", subbytes_output);
        $display("First 4 bytes mapping:");
        $display("  0x00 -> 0x%h (Expected: 0x63)", subbytes_output[7:0]);
        $display("  0x01 -> 0x%h (Expected: 0x7c)", subbytes_output[15:8]);
        $display("  0x02 -> 0x%h (Expected: 0x77)", subbytes_output[23:16]);
        $display("  0x03 -> 0x%h (Expected: 0x7b)", subbytes_output[31:24]);
        
        // TEST 5: Alternating 0xAA
        $display("\n\nTEST 5: All 0xAA Pattern");
        test_state = 128'haaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa;
        $display("Input State:  %h", test_state);
        
        #2;
        
        $display("Output State: %h", subbytes_output);
        $display("Explanation:  sbox[0xaa] = 0x%h for all bytes", subbytes_output[7:0]);
        
        // TEST 6: Alternating 0x55
        $display("\n\nTEST 6: All 0x55 Pattern");
        test_state = 128'h55555555555555555555555555555555;
        $display("Input State:  %h", test_state);
        
        #2;
        
        $display("Output State: %h", subbytes_output);
        $display("Explanation:  sbox[0x55] = 0x%h for all bytes", subbytes_output[7:0]);
        
        // TEST 7: Mixed Pattern
        $display("\n\nTEST 7: Mixed Pattern");
        test_state = 128'h0123456789abcdef0123456789abcdef;
        $display("Input State:  %h", test_state);
        
        #2;
        
        $display("Output State: %h", subbytes_output);
        
        // TEST 8: Check specific S-box entries
        $display("\n\nTEST 8: Specific S-box Entry Verification");
        test_state = 128'h53000000000000000000000000000000;
        $display("Input State:  %h", test_state);
        
        #2;
        
        $display("Output State: %h", subbytes_output);
        $display("sbox[0x53] = 0x%h (Expected: 0xed)", subbytes_output[7:0]);
        
        if (subbytes_output[7:0] == 8'hed)
            $display("? TEST 8 PASSED");
        else
            $display("? TEST 8 FAILED");
        
        // TEST 9: Boundary values
        $display("\n\nTEST 9: Boundary Values");
        test_state = 128'h01fe80007f0000000000000000000000;
        $display("Input State:  %h", test_state);
        
        #2;
        
        $display("Output State: %h", subbytes_output);
        $display("Byte mappings:");
        $display("  sbox[0x01] = 0x%h", subbytes_output[7:0]);
        $display("  sbox[0xfe] = 0x%h", subbytes_output[15:8]);
        $display("  sbox[0x80] = 0x%h", subbytes_output[23:16]);
        $display("  sbox[0x7f] = 0x%h", subbytes_output[31:24]);
        
        $display("\n========== SUBBYTES TEST COMPLETE ==========\n");
        $finish;
    end

endmodule
