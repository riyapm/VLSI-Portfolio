module ShiftRows (
    input  [127:0] state_in,   // 128-bit input state
    output [127:0] state_out   // 128-bit output state
);

    // AES state is organized as a 4x4 matrix of bytes
    // Byte ordering: state_in[127:120] = byte 0, state_in[119:112] = byte 1, etc.
    // Matrix layout (column-major in AES):
    //   [0]  [4]  [8]  [12]
    //   [1]  [5]  [9]  [13]
    //   [2]  [6]  [10] [14]
    //   [3]  [7]  [11] [15]

    // ShiftRows operation:
    // Row 0: No shift
    // Row 1: Shift left by 1 byte
    // Row 2: Shift left by 2 bytes
    // Row 3: Shift left by 3 bytes

    // Row 0 (bytes 0, 4, 8, 12) - no shift
    assign state_out[127:120] = state_in[127:120]; // byte 0
    assign state_out[95:88]   = state_in[95:88];   // byte 4
    assign state_out[63:56]   = state_in[63:56];   // byte 8
    assign state_out[31:24]   = state_in[31:24];   // byte 12

    // Row 1 (bytes 1, 5, 9, 13) - shift left by 1
    assign state_out[119:112] = state_in[87:80];   // byte 1 <- byte 5
    assign state_out[87:80]   = state_in[55:48];   // byte 5 <- byte 9
    assign state_out[55:48]   = state_in[23:16];   // byte 9 <- byte 13
    assign state_out[23:16]   = state_in[119:112]; // byte 13 <- byte 1

    // Row 2 (bytes 2, 6, 10, 14) - shift left by 2
    assign state_out[111:104] = state_in[47:40];   // byte 2 <- byte 10
    assign state_out[79:72]   = state_in[15:8];    // byte 6 <- byte 14
    assign state_out[47:40]   = state_in[111:104]; // byte 10 <- byte 2
    assign state_out[15:8]    = state_in[79:72];   // byte 14 <- byte 6

    // Row 3 (bytes 3, 7, 11, 15) - shift left by 3
    assign state_out[103:96]  = state_in[7:0];     // byte 3 <- byte 15
    assign state_out[71:64]   = state_in[103:96];  // byte 7 <- byte 3
    assign state_out[39:32]   = state_in[71:64];   // byte 11 <- byte 7
    assign state_out[7:0]     = state_in[39:32];   // byte 15 <- byte 11

endmodule
