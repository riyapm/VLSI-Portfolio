module tb_shiftrows;

    reg [127:0] test_state;
    wire [127:0] shifted_state;

    ShiftRows uut (
        .state_in(test_state),
        .state_out(shifted_state)
    );

    initial begin
        $display("========== ShiftRows Test ==========");
        
        // Test 1: Known pattern
        test_state = 128'h00112233445566778899aabbccddeeff;
        #1;
        $display("Input:  %h", test_state);
        $display("Output: %h", shifted_state);
        $display("Expected: 0055aaff4499ee33881cddb2776cc11");
        
        // Test 2: All zeros
        test_state = 128'h00000000000000000000000000000000;
        #1;
        $display("\nInput:  %h", test_state);
        $display("Output: %h", shifted_state);
        $display("Expected: all zeros");
        
        // Test 3: Sequential bytes
        test_state = 128'h000102030405060708090a0b0c0d0e0f;
        #1;
        $display("\nInput:  %h", test_state);
        $display("Output: %h", shifted_state);
        
        $finish;
    end

endmodule
